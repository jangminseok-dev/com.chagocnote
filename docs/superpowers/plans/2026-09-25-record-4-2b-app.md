**Goal:** 기록에 붙은 사진을 보는 쪽을 만든다 — 방문 상세의 기울어진 사진 넘기기와 크게 보기(원본 저장·전/후 바꾸기·대표로·삭제), 캘린더 탭의 [달력 / 피드] 토글과 피드, 달력 칸의 대표 사진과 미리보기의 부채꼴, 하루 목록 카드의 사진, 고객 탭의 프로필 사진과 반려동물 상세의 [기록] 탭.

## 이 계획에서 새로 정한 것

설계와 다른 것과 새로 정한 것은 결정 기록 007 의 "④-2b 앱(보기)에서 정한 것" 절(작업 9)에 적는다. 요약:
- "대표로"는 그 칸(전/후)의 맨 앞으로 옮기고, 실제로 대표가 되는 사진에만 보인다(후 사진이 있으면 전 사진은 대표가 아니다).
- 방문 상세의 이름 카드는 불투명이다(스크롤 화면). 피드의 이름 카드만 글래스이고, 실기에서 비쳐 보여 품종·몸무게 줄을 잉크색으로 쓴다.
- 하루 목록 카드의 사진 자리도 채운다. 달력 미리보기는 줄 목록 위에 부채꼴을 얹는다(줄은 남긴다). 캘린더 탭 머리는 두 줄([달력/피드] 토글·설정 / 월).
- 크게 보기는 원본을 화면에 맞춰 보이고 두 손가락으로 키운다. 영상은 포스터 + 재생 표시만(④-3). 보던 사진은 ID 로 따라가고, 마지막 사진을 지우면 닫고, 되돌린 방문은 원본 저장만.
- `buildRoot({overrides})` — 시뮬레이터 흐름이 사진 선택기를 바꿔 끼우며 실제 조립(플러그 꽂기)을 쓴다. 흐름 공용 도구는 `integration_test/flow_kit.dart`.

  // 가운데 카드(162)가 화면 폭의 반쯤이라 양옆 카드가 보인다.

    super.didUpdateWidget(old);

    if (_index >= widget.photos.length && widget.photos.isNotEmpty) {

  void _tap(int i) {
    if (i == _index) {
      widget.onTap?.call(i);
      return;
    }
    _controller.animateToPage(
      i,
      duration: const Duration(milliseconds: 280),
      curve: Curves.easeOut,

/// 크게 보기에서 사진 하나를 바꾸는 규칙(설계 10.1 ⑤) — 전/후 바꾸기·대표로·빼기는 기록 전체 교체 한 번이다.
library;

import '../api/record_models.dart';
import 'record_rules.dart';

enum PhotoEdit { togglePhase, makeCover, remove }

/// 넘기기·크게 보기에서 보이는 순서 — 서버의 대표 규칙과 같다(후 사진 → 전 사진 → 영상, 칸마다 순서대로).
/// 그래서 첫 장이 늘 대표 사진이다.
List<RecordMedia> galleryOrder(List<RecordMedia> media) =>
    [...media]..sort((a, b) {
      int rank(RecordMedia m) =>
          (m.kind == MediaKind.photo ? 0 : 2) +
          (m.phase == MediaPhase.after ? 0 : 1);
      final byRank = rank(a).compareTo(rank(b));
      return byRank != 0 ? byRank : a.sortOrder.compareTo(b.sortOrder);
    });

/// "대표로"를 보일지 — 그 칸 맨 앞으로 옮겨 실제로 대표가 되는 사진에만. 후 사진이 있으면 전 사진은
/// 맨 앞으로 옮겨도 대표가 아니고, 사진이 있으면 영상도 대표가 아니다.
bool canMakeCover(List<RecordMedia> media, int mediaId) {
  final target = media.where((m) => m.mediaId == mediaId).firstOrNull;
  if (target == null || target.kind != MediaKind.photo) return false;
  if (coverOf(media)?.mediaId == mediaId) return false;
  return target.phase == MediaPhase.after ||
      !media.any(
        (m) => m.kind == MediaKind.photo && m.phase == MediaPhase.after,
      );
}

/// 바꾼 뒤의 사진 자리. 그 사진이 그사이 빠졌으면 null(보낼 것이 없다).
List<MediaPlacement>? applyPhotoEdit(
  List<RecordMedia> media,
  int mediaId,
  PhotoEdit edit,
) {
  final target = media.where((m) => m.mediaId == mediaId).firstOrNull;
  if (target == null) return null;
  // 서버가 주는 순서(전 → 후, 칸마다 순서) 그대로 두고 그 사진만 옮긴다.
  final ordered = [
    for (final m in [...media]..sort(_byPhaseThenOrder))
      if (m.mediaId != mediaId) FormMedia(m, m.phase),
  ];
  switch (edit) {
    case PhotoEdit.remove:
      break;
    case PhotoEdit.togglePhase:
      ordered.add(
        FormMedia(
          target,
          target.phase == MediaPhase.before
              ? MediaPhase.after
              : MediaPhase.before,
        ),
      );
    case PhotoEdit.makeCover:
      final first = ordered.indexWhere((m) => m.phase == target.phase);
      ordered.insert(
        first < 0 ? ordered.length : first,
        FormMedia(target, target.phase),
      );
  }
  return placementsOf(ordered);
}

int _byPhaseThenOrder(RecordMedia a, RecordMedia b) {
  final byPhase = a.phase.index.compareTo(b.phase.index);
  return byPhase != 0 ? byPhase : a.sortOrder.compareTo(b.sortOrder);
}

/// 사진 자리만 바꾼 전체 교체 — 나머지 칸과 버전은 읽은 기록 그대로 보낸다.
RecordInput recordInputOf(VisitRecord record, List<MediaPlacement> media) =>
    RecordInput(
      version: record.version,
      weightKg: record.weightKg,
      careNote: record.careNote,
      internalNote: record.internalNote,
      paymentMethod: record.paymentMethod,

class FeedState {
  const FeedState({
    required this.items,
    required this.nextCursor,
    this.loadingMore = false,
    this.loadMoreError,
  });

  final List<FeedItem> items;
  final String? nextCursor;

  /// 다음 쪽을 불러오는 중.
  final bool loadingMore;

  /// 마지막 다음 쪽 요청의 오류. 이미 보인 목록은 그대로 둔다.
  final Object? loadMoreError;

  bool get hasMore => nextCursor != null;
}

/// 완료 방문 피드 — 커서로 이어 읽는다(설계 9절). 기록·전이를 쓰면 `VisitCommands`가 무효화한다.
class FeedController extends AsyncNotifier<FeedState> {
  FeedController(this.query);

  final FeedQuery query;

  /// build() 가 돌 때마다 하나씩 는다 — 다시 불러오는 사이 끝난 다음 쪽 요청이 새 목록에 옛 쪽을
  /// 붙이지 않게 비교한다(고객 목록과 같은 이유).
  int _generation = 0;

  Future<VisitFeed> _fetch(String? cursor) => runShopScoped(
    ref,
    () => ref
        .read(visitApiProvider)
        .feed(query.shopId, cursor: cursor, petId: query.petId),
  );

  @override
  Future<FeedState> build() async {
    _generation++;
    final page = await _fetch(null);
    return FeedState(items: page.items, nextCursor: page.nextCursor);
  }

  /// 다음 쪽을 붙인다. 첫 쪽을 불러오는 중이거나, 이미 다음 쪽을 불러오는 중이거나, 마지막 쪽이면
  /// 아무것도 하지 않는다.
  Future<void> loadMore() async {
    final current = state.value;
    if (state.isLoading ||
        current == null ||
        !current.hasMore ||
        current.loadingMore) {
      return;
    }
    final generation = _generation;
    bool stale() => !ref.mounted || generation != _generation;
    state = AsyncData(
      FeedState(
        items: current.items,
        nextCursor: current.nextCursor,
        loadingMore: true,
      ),
    );
    try {
      final page = await _fetch(current.nextCursor);
      if (stale()) return;
      state = AsyncData(
        FeedState(
          items: [...current.items, ...page.items],
          nextCursor: page.nextCursor,
        ),
      );
    } catch (e) {
      if (stale()) return;
      state = AsyncData(
        FeedState(
          items: current.items,
          nextCursor: current.nextCursor,
          loadMoreError: e,
        ),
      );
    }
  }
}

Future<List<int>> download(String url) async {
  final client = HttpClient();
  try {
    final response = await (await client.getUrl(Uri.parse(url))).close();
    final bytes = <int>[];
    await for (final chunk in response) {
      bytes.addAll(chunk);
    }
    return bytes;
  } finally {
    client.close();
  }
}

  final recorder = ui.PictureRecorder();
  final canvas = ui.Canvas(recorder);
  canvas.drawRect(
    const ui.Rect.fromLTWH(0, 0, 4000, 3000),
    ui.Paint()..color = ui.Color(0xFF000000 + 0x303030 * (shade % 4 + 1)),
  );
  canvas.drawCircle(
    const ui.Offset(2000, 1500),
    900,
    ui.Paint()..color = const ui.Color(0xFFF0F0F0),
  );
  final image = await recorder.endRecording().toImage(4000, 3000);
  final png = await image.toByteData(format: ui.ImageByteFormat.png);
  final file = File('${Directory.systemTemp.path}/$name.png');
  await file.writeAsBytes(png!.buffer.asUint8List());
  return file.path;
}
